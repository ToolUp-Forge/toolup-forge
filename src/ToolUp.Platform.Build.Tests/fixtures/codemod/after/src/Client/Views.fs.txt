module MyApp.Views

open Feliz
open ToolUp.Platform
open Toolup.UIToolkit

// Phase 1075 — one view contract, the toolkit's locale, one shell entry.
let sales () =
    ClientModule.create { Init = init; Update = update; Name = "Sales"; Icon = Html.none }
    |> ClientModule.withView (fun model ctx -> PageContent.SplitPanel(view model ctx.Dispatch))
    |> ClientModule.register

let settings () =
    ClientModule.create { Init = init; Update = update; Name = "Settings"; Icon = Html.none }
    |> ClientModule.withView (fun model ctx -> PageContent.FullWidth(settingsView model ctx.Dispatch))
    |> ClientModule.register

let reports () =
    ClientModule.create { Init = init; Update = update; Name = "Reports"; Icon = Html.none }
    |> ClientModule.withPages [ overviewPage, overviewView ]
    |> ClientModule.register

let sections = Toolup.Sidebar.buildSections catalog.Sidebar modules prefs
let degraded = BootDegradation.banner catalog.BootDegradation degradations retry dismiss
let total = OutputFormatting.formatDecimal amount
let shell = Layout.AppShellWith sidebar content None None Narrow
let heading = (MessageCatalogProvider.useMessages ()).Shell.Loading
