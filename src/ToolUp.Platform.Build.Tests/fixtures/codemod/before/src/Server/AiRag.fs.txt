module MyApp.Server.AiRag

open ToolUp.Platform
open ToolUp.AI.AICompose
open ToolUp.RAG.RAGCompose

/// An AI root on the wrapper record.
let ai factory profile (config: ServerConfig) =
    AIServerApp.create factory profile
    |> AIServerApp.withConfig config
    |> AIServerApp.run

/// A RAG root on the wrapper record.
let rag factory profile embedder (config: ServerConfig) =
    RAGServerApp.create factory profile embedder
    |> RAGServerApp.withConfig config
    |> RAGServerApp.run

/// The additive extensions, in their configurator shape.
let stacked factory profile embedder (app: ServerApp) =
    app
    |> withAI factory profile id
    |> withRAG factory profile embedder id
