module Server.Infrastructure

open ToolUp.Platform
open ToolUp.Platform.Jobs

let record (eventStore: IEventStore) evt = async {
    do! eventStore.Write evt
}

let save (jobStore: IJobStore) job = async {
    do! jobStore.Save job
}

let retire (scheduler: IJobScheduler) scopeId jobId = async {
    do! scheduler.Cancel(scopeId, jobId)
}

let persist (store: IEventStore) (e: ModuleEvent) = async {
    let! written = store.Write(e)
    return written
}

let update (store: IJobStore) job = async {
    do! store.Update job
}

type Recorder() =
    interface IEventStore with
        member _.Write(evt) = async { return Ok() }

type Retirer() =
    member _.Cancel(scopeId, jobId) = async { return Ok() }

let audited (auditLog: IAuditLog) scopeId event = async {
    do! auditLog.Record(scopeId, event)
}

let gate registry (auditLog: IAuditLog) =
    GrantPolicyGuard.guardDispatch registry Map.empty (Some auditLog) Async.Start "t1" "alice" "Reports"

type QuietLog() =
    interface IAuditLog with
        member _.Record(scopeId, audit) = async { return () }
        member _.GetAuditTrail(_, _, _) = async { return [] }

type FixedPermissions() =
    interface IPermissionStore with
        member _.GetTeamPermissions teamId = async { return TeamPermissions.empty }

let envelope scopeId at event = AuditEnvelope.fromScopeId scopeId at event

let sink (audit: IAuditLog option) = audit

let quiet auditLog = match auditLog with Some log -> log | None -> NoOpAuditLog() :> IAuditLog

let recordSubmission (log: IAuditLog) teamId actor payload =
    AuditRow.emit log (AuditScope.team teamId) actor (AuditEvent.FormSubmitted payload)

let readArchiveLine (line: string) =
    JsonSerializer.Deserialize<ArchiveLine>(line, PlatformJson.options)

let fallback = AccessContext.unrestricted (AnonymousSession "anonymous")
let jobContext subject = AccessContext.unrestricted subject

let isOpen (ctx: AccessContext) = ctx.ModulePermissions.IsEmpty

let stamp (ctx: HttpContext) perms = ctx.Items["ToolUp.ModulePermissions"] <- box perms
