module MyApp.Server.Program

open Giraffe
open Microsoft.AspNetCore.Http
open Fable.Remoting.Server
open Fable.Remoting.Giraffe
open Fable.Remoting.AspNetCore
open ToolUp.Platform
open ToolUp.Platform.Server

// The public-submit route used to ride an anonymous prefix that
// AuthEnforcementMiddleware honoured.
let config = {
    ServerConfig.defaults with
        Mode = Individual
        AcceptHeaderAuthInAuthenticatedMode = true
        AcceptPlaintextSecretsInAuthenticatedMode = true
        RateLimit = Some RateLimitPolicy.defaults
}

let scopePath (ctx: AccessContext) =
    match ctx.Mode with
    | Team
    | MultiTeam -> sprintf "team-%s" ctx.TeamId.Value
    | _ -> sprintf "user-%s" ctx.UserId

let isPublic (ctx: AccessContext) = ctx.Mode = Anonymous

let submitHandler (config: ServerConfig) (mode: PlatformMode) : HttpHandler = fun next ctx -> next ctx

let routeInfo: Fable.Remoting.Server.RouteInfo<HttpContext> option = None

let app =
    ServerApp.empty
    |> ServerApp.withConfig config
    |> ServerApp.withAnonymousRoute "/api/forms/public/"

[<EntryPoint>]
let main argv =
    Environment.SetEnvironmentVariable("TOOLUP_PLATFORM_MODE", "individual")
    ServerApp.run app
    0
