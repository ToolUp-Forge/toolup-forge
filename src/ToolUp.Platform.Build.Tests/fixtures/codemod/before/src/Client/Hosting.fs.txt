module MyApp.Client.Hosting

open ToolUp.Platform

let activeState (model: Client.Model) =
    model.ModuleStates |> Map.tryFind model.ActiveModuleId

let empty: Client.Model = {
    ModuleStates = Map.empty
    ConfigsPrefetch = Remote.Loading
    FlagsPrefetch = Remote.Loading
    ResetCounters = Map.empty
}

let increment dispatch = dispatch (Client.ModuleMsg(box Increment))
